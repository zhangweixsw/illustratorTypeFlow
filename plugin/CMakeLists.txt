cmake_minimum_required(VERSION 3.24)
project(IllustratorTypeFlowPlugin LANGUAGES CXX)

if(NOT DEFINED ILLUSTRATOR_SDK_ROOT)
  message(FATAL_ERROR "Pass -DILLUSTRATOR_SDK_ROOT=<Adobe Illustrator 2026 SDK directory>")
endif()

set(AI_API "${ILLUSTRATOR_SDK_ROOT}/illustratorapi")
foreach(required
    "${AI_API}/IllustratorSDK.h"
    "${AI_API}/illustrator/AIDocument.h"
    "${AI_API}/illustrator/AINotifier.h")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Required SDK file not found: ${required}")
  endif()
endforeach()

add_library(IllustratorTypeFlowPlugin MODULE
  src/PluginMain.cpp
  src/PipePublisher.cpp
  src/PipePublisher.h
  IllustratorTypeFlowPlugin.def)

target_compile_features(IllustratorTypeFlowPlugin PRIVATE cxx_std_20)
target_compile_definitions(IllustratorTypeFlowPlugin PRIVATE
  WIN_ENV=1
  _WINDOWS=1
  WIN32_LEAN_AND_MEAN=1
  NOMINMAX=1)
target_compile_options(IllustratorTypeFlowPlugin PRIVATE /EHsc /utf-8 /W4)
target_include_directories(IllustratorTypeFlowPlugin PRIVATE
  "${AI_API}"
  "${AI_API}/illustrator"
  "${AI_API}/ate"
  "${AI_API}/pica_sp")
target_link_libraries(IllustratorTypeFlowPlugin PRIVATE user32)

set_target_properties(IllustratorTypeFlowPlugin PROPERTIES
  PREFIX ""
  SUFFIX ".aip"
  OUTPUT_NAME "IllustratorTypeFlow")

# Illustrator 2026 requires a PiPL generated by the proprietary SDK tooling.
# scripts/build-plugin.ps1 runs create_pipl.py and places plugin.pipl beside the AIP.
